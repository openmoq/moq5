cmake_minimum_required(VERSION 3.20)
project(moq_cold_component_producer LANGUAGES C)
include(GNUInstallDirs)
foreach(option MOQ_BUILD_TESTS MOQ_BUILD_EXAMPLES MOQ_BUILD_SIM MOQ_BUILD_SERVICE
        MOQ_BUILD_CPP_BINDING MOQ_BUILD_RUST_BINDING MOQ_BUILD_LOC MOQ_BUILD_CMAF
        MOQ_BUILD_MSF MOQ_BUILD_CODEC_SIGNALING MOQ_BUILD_CMAF_PACKAGER
        MOQ_BUILD_MEDIA_OBJECT MOQ_BUILD_MPEG2TS MOQ_BUILD_PLAYBACK MOQ_BUILD_RELAY
        MOQ_BUILD_RELAY_CORE MOQ_BUILD_RELAY_RUNTIME MOQ_BUILD_RELAY_TOOL
        MOQ_BUILD_ADAPTER_PICOQUIC MOQ_BUILD_PQ_THREADED MOQ_BUILD_ADAPTER_PICO_WT
        MOQ_BUILD_PICO_WT_MANAGED MOQ_BUILD_ADAPTER_MSQUIC MOQ_BUILD_MSQUIC_MANAGED
        MOQ_BUILD_ADAPTER_WTQUIC MOQ_BUILD_WTQUIC_NETWORK_MANAGED
        MOQ_BUILD_WTQUIC_MSQUIC_MANAGED MOQ_BUILD_ADAPTER_MVFST MOQ_BUILD_ADAPTER_PROXYGEN)
    set(${option} OFF CACHE BOOL "Cold component producer scope" FORCE)
endforeach()
set(MOQ_BUILD_PACKAGE_EXPORTS ON CACHE BOOL "Qualify native packages" FORCE)
if(MOQ_COLD_COMPONENT STREQUAL "adapter-msquic")
    set(MOQ_BUILD_ADAPTER_MSQUIC ON CACHE BOOL "Actual MsQuic SDK adapter" FORCE)
    set(MOQ_BUILD_MSQUIC_MANAGED "${FIXTURE_MANAGED}" CACHE BOOL "Compiled producer closure" FORCE)
elseif(MOQ_COLD_COMPONENT STREQUAL "adapter-wtquic-network-managed" AND APPLE)
    set(MOQ_BUILD_ADAPTER_WTQUIC ON CACHE BOOL "Actual wtquic SDK adapter" FORCE)
    set(MOQ_BUILD_WTQUIC_NETWORK_MANAGED ON CACHE BOOL "Actual Network SDK facade" FORCE)
else()
    message(FATAL_ERROR "Cold component requires actual MsQuic or Apple wtquic Network SDK")
endif()
add_subdirectory("${MOQ_SOURCE}" libmoq)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cold-expected.cmake" CONTENT
    "set(expected [==[$<TARGET_FILE:moq::${MOQ_COLD_COMPONENT}>]==])\nset(expected_kind [==[$<TARGET_PROPERTY:moq::${MOQ_COLD_COMPONENT},TYPE>]==])\nset(package_libdir [==[${CMAKE_INSTALL_LIBDIR}]==])\n")
