cmake_minimum_required(VERSION 3.20)
project(moq_manual_export_producer LANGUAGES C CXX)
get_property(multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(multi)
    if(MOQ_RELEASE_ONLY)
        set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Fixture configurations" FORCE)
    else()
        set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Fixture configurations" FORCE)
    endif()
endif()
if(WIN32)
    message(FATAL_ERROR "This fixture qualifies POSIX imports, not native Windows")
endif()
foreach(option MOQ_BUILD_TESTS MOQ_BUILD_EXAMPLES MOQ_BUILD_SIM MOQ_BUILD_RELAY
        MOQ_BUILD_RELAY_CORE MOQ_BUILD_RELAY_RUNTIME MOQ_BUILD_RELAY_TOOL
        MOQ_BUILD_ADAPTER_PICOQUIC MOQ_BUILD_PQ_THREADED MOQ_BUILD_ADAPTER_PICO_WT
        MOQ_BUILD_PICO_WT_MANAGED MOQ_BUILD_ADAPTER_MSQUIC MOQ_BUILD_MSQUIC_MANAGED
        MOQ_BUILD_ADAPTER_WTQUIC MOQ_BUILD_WTQUIC_NETWORK_MANAGED
        MOQ_BUILD_WTQUIC_MSQUIC_MANAGED MOQ_BUILD_ADAPTER_MVFST MOQ_BUILD_ADAPTER_PROXYGEN)
    set(${option} OFF CACHE BOOL "Test-owned producer selection" FORCE)
endforeach()
foreach(option MOQ_BUILD_SERVICE MOQ_BUILD_MSF MOQ_BUILD_LOC MOQ_BUILD_CMAF
        MOQ_BUILD_MEDIA_OBJECT MOQ_WARNINGS_AS_ERRORS)
    set(${option} ON CACHE BOOL "Test-owned producer selection" FORCE)
endforeach()
if(MOQ_MODE STREQUAL "embedding")
    set(MOQ_BUILD_CPP_BINDING ON CACHE BOOL "Embedded public C++ target" FORCE)
    set(MOQ_BUILD_RELAY_CORE ON CACHE BOOL "Embedded relay exports" FORCE)
    set(MOQ_BUILD_RELAY_RUNTIME ON CACHE BOOL "Embedded relay exports" FORCE)
    if(DEFINED FIXTURE_SERVICE AND NOT FIXTURE_SERVICE)
        set(MOQ_BUILD_SERVICE OFF CACHE BOOL "Disabled embedding service" FORCE)
    endif()
endif()
if(MOQ_MODE MATCHES "^pico")
    if(NOT MOQ_PICOQUIC_SOURCE_DIR)
        message(FATAL_ERROR "PicoQUIC evidence requires a real source graph")
    endif()
    set(MOQ_BUILD_ADAPTER_PICOQUIC ON CACHE BOOL "Real adapter" FORCE)
    set(MOQ_BUILD_PQ_THREADED ON CACHE BOOL "Real managed facade" FORCE)
endif()
if(MOQ_MODE STREQUAL "pico-wt")
    set(MOQ_BUILD_ADAPTER_PICO_WT ON CACHE BOOL "Real WT adapter" FORCE)
    set(MOQ_BUILD_PICO_WT_MANAGED ON CACHE BOOL "Real WT managed facade" FORCE)
endif()
add_subdirectory("${MOQ_SOURCE}" libmoq)
set(components "")
set(targets moq::core moq::msf)
set(modules "")
if(TARGET moq::service)
    list(APPEND components service)
    list(APPEND targets moq::service)
    list(APPEND modules libmoqServiceTargets)
endif()
if(MOQ_MODE MATCHES "^pico")
    list(APPEND components adapter-picoquic adapter-picoquic-threaded)
    list(APPEND targets moq::adapter-picoquic moq::adapter-picoquic-threaded)
    list(APPEND modules libmoqAdapterTargets libmoqAdapterThreadedTargets)
endif()
if(MOQ_MODE STREQUAL "pico-wt")
    list(APPEND components adapter-pico-wt adapter-pico-wt-managed)
    list(APPEND targets moq::adapter-pico-wt moq::adapter-pico-wt-managed)
    list(APPEND modules libmoqPicoWtAdapterTargets libmoqPicoWtManagedTargets)
endif()
# Force different configured filenames as well as directories. This also
# exercises selected-config install modules instead of the old fixed output.
foreach(component IN LISTS components)
    if(component STREQUAL "service")
        set(real moq-service)
    else()
        set(real "moq-${component}")
    endif()
    set_target_properties(${real} PROPERTIES
        OUTPUT_NAME_DEBUG "${real}-debug" OUTPUT_NAME_RELEASE "${real}-release")
endforeach()
set(metadata "set(components [==[${components}]==])\nset(targets [==[${targets}]==])\nset(modules [==[${modules}]==])\n")
foreach(target IN LISTS targets)
    string(MAKE_C_IDENTIFIER "${target}" id)
    string(APPEND metadata "set(expected_${id} [==[$<TARGET_FILE:${target}>]==])\nset(kind_${id} [==[$<TARGET_PROPERTY:${target},TYPE>]==])\n")
endforeach()
if(MOQ_MODE MATCHES "^pico")
    set(provider_targets picoquic::picoquic-core picoquic::picoquic-log)
    if(MOQ_MODE STREQUAL "pico-wt")
        list(APPEND provider_targets picoquic::picohttp-core)
    endif()
    string(APPEND metadata "set(provider_targets [==[${provider_targets}]==])\n")
    foreach(target IN LISTS provider_targets)
        string(MAKE_C_IDENTIFIER "${target}" id)
        string(APPEND metadata "set(provider_expected_${id} [==[$<TARGET_FILE:${target}>]==])\n")
    endforeach()
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/expected-$<IF:$<BOOL:$<CONFIG>>,$<CONFIG>,NOCONFIG>.cmake"
    CONTENT "${metadata}")
file(WRITE "${CMAKE_BINARY_DIR}/package.cmake"
    "set(package_name libmoq)\nset(package_dir [==[${CMAKE_CURRENT_BINARY_DIR}/libmoq]==])\nset(package_libdir [==[${CMAKE_INSTALL_LIBDIR}]==])\n")
