cmake_minimum_required(VERSION 3.20)
project(wtquic_api_consumer LANGUAGES ${PROBE_LANGUAGE})
if(LOOKUP STREQUAL "optional-component")
    find_package(libmoq REQUIRED CONFIG OPTIONAL_COMPONENTS ${COMPONENT})
    if(NOT libmoq_FOUND OR libmoq_${COMPONENT}_FOUND OR TARGET moq::${COMPONENT})
        message(FATAL_ERROR "optional component was not cleanly unavailable")
    endif()
    if(NOT libmoq_NOT_FOUND_MESSAGE MATCHES "wtquic public API compile/link contract requires")
        message(FATAL_ERROR "missing optional component diagnostic")
    endif()
    return()
elseif(LOOKUP STREQUAL "quiet")
    find_package(libmoq QUIET CONFIG COMPONENTS ${COMPONENT})
    if(libmoq_FOUND OR TARGET moq::${COMPONENT})
        message(FATAL_ERROR "incompatible package was accepted")
    endif()
    if(NOT libmoq_NOT_FOUND_MESSAGE MATCHES "wtquic public API compile/link contract requires")
        message(FATAL_ERROR "missing optional package diagnostic")
    endif()
    return()
endif()
find_package(libmoq REQUIRED CONFIG COMPONENTS ${COMPONENT})
if(COMPONENT STREQUAL "service")
    set(header endpoint)
    set(type moq_endpoint_cfg_t)
    set(init moq_endpoint_cfg_init_sized)
elseif(COMPONENT STREQUAL "adapter-wtquic")
    set(header wtquic)
    set(type moq_wtquic_conn_cfg_t)
    set(init moq_wtquic_conn_cfg_init_sized)
else()
    string(REPLACE "adapter-" "" header "${COMPONENT}")
    string(REPLACE "-" "_" header "${header}")
    set(type "moq_${header}_cfg_t")
    set(init "moq_${header}_cfg_init_sized")
endif()
if(PROBE_LANGUAGE STREQUAL "CXX")
    set(ext cpp)
else()
    set(ext c)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.${ext}"
    "#include <moq/${header}.h>\nint main(void) { ${type} cfg; ${init}(&cfg, sizeof(cfg)); return 0; }\n")
add_executable(consumer "${CMAKE_CURRENT_BINARY_DIR}/main.${ext}")
target_link_libraries(consumer PRIVATE moq::${COMPONENT})
