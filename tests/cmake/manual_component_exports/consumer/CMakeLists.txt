cmake_minimum_required(VERSION 3.20)
project(moq_component_export_consumer LANGUAGES C CXX)
if(MOQ_CONSUMER_CASE STREQUAL "mapping")
    # Prefer Release for configured imports, while retaining configless system
    # dependencies such as FindOpenSSL's imported targets.
    set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO "Release;")
endif()
get_property(multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(multi)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Fixture configurations" FORCE)
endif()
if(MOQ_CONSUMER_CASE STREQUAL "empty-release")
    set(MOQ_CONFIGS Release)
elseif(MOQ_CONSUMER_CASE STREQUAL "mapping")
    set(MOQ_CONFIGS Debug Release)
elseif(MOQ_INSTALLED)
    set(MOQ_CONFIGS "${MOQ_FIRST_CONFIG}")
elseif(multi)
    set(MOQ_CONFIGS Debug Release)
else()
    set(MOQ_CONFIGS NOCONFIG)
endif()
include("${MOQ_EXPECTED}/expected-${MOQ_FIRST_CONFIG}.cmake")
if(package_name STREQUAL "libmoq")
    if(MOQ_CONSUMER_CASE STREQUAL "service-only")
        # No explicit facade/component finds in this cold consumer scope.
        find_package(libmoq CONFIG REQUIRED COMPONENTS service)
        find_package(libmoq CONFIG REQUIRED COMPONENTS service)
    else()
        find_package(libmoq CONFIG REQUIRED COMPONENTS ${components})
        find_package(libmoq CONFIG REQUIRED COMPONENTS ${components})
    endif()
else()
    find_package(component_test CONFIG REQUIRED)
    find_package(component_test CONFIG REQUIRED)
endif()
if(MOQ_CONSUMER_CASE STREQUAL "mapping")
    foreach(component IN LISTS components)
        get_target_property(mapping "moq::${component}" MAP_IMPORTED_CONFIG_RELWITHDEBINFO)
        if(NOT mapping STREQUAL "Release;")
            message(FATAL_ERROR "Application import mapping was overwritten: ${component}/${mapping}")
        endif()
    endforeach()
endif()
# Check the stored properties themselves: a tolerant configure or a link that
# happens to find the first imported configuration cannot satisfy this oracle.
foreach(config IN LISTS MOQ_CONFIGS)
    include("${MOQ_EXPECTED}/expected-${config}.cmake")
    string(TOUPPER "${config}" upper)
    if(NOT MOQ_INSTALLED)
        foreach(target IN LISTS provider_targets)
            string(MAKE_C_IDENTIFIER "${target}" id)
            get_target_property(imported "${target}" IMPORTED)
            get_target_property(location "${target}" "IMPORTED_LOCATION_${upper}")
            if(NOT imported OR NOT location STREQUAL provider_expected_${id})
                message(FATAL_ERROR "Build-tree provider was rebuilt/reselected: ${target}/${config}/${location}")
            endif()
        endforeach()
    endif()
    foreach(target IN LISTS targets)
        string(MAKE_C_IDENTIFIER "${target}" id)
        if(NOT TARGET "${target}")
            message(FATAL_ERROR "Package did not autoload required owned target ${target}")
        endif()
        get_target_property(kind "${target}" TYPE)
        if(NOT kind STREQUAL kind_${id})
            message(FATAL_ERROR "configuration artifact identity: wrong kind for ${target}")
        endif()
        get_target_property(declared "${target}" IMPORTED_CONFIGURATIONS)
        if(MOQ_INSTALLED OR MOQ_CONSUMER_CASE STREQUAL "empty-release")
            if(NOT declared STREQUAL upper)
                message(FATAL_ERROR "Stale configuration record for ${target}: ${declared} != ${upper}")
            endif()
        endif()
        if(MOQ_CONSUMER_CASE STREQUAL "empty-release")
            get_property(stale_debug TARGET "${target}" PROPERTY IMPORTED_LOCATION_DEBUG SET)
            if(stale_debug)
                message(FATAL_ERROR "Stale Debug artifact survived producer reconfiguration: ${target}")
            endif()
        endif()
        get_target_property(location "${target}" "IMPORTED_LOCATION_${upper}")
        if(MOQ_INSTALLED)
            if(NOT MOQ_INSTALL_PREFIX OR NOT DEFINED MOQ_INSTALL_LIBDIR OR
               IS_ABSOLUTE "${MOQ_INSTALL_LIBDIR}")
                message(FATAL_ERROR "Installed fixture requires its owned prefix and relative libdir")
            endif()
            file(REAL_PATH "${MOQ_INSTALL_PREFIX}" prefix_real)
            get_filename_component(filename "${expected_${id}}" NAME)
            set(owned "${prefix_real}/${MOQ_INSTALL_LIBDIR}/${filename}")
            cmake_path(NORMAL_PATH owned)
            get_filename_component(location_absolute "${location}" ABSOLUTE)
            if(NOT location_absolute STREQUAL owned OR NOT EXISTS "${owned}")
                message(FATAL_ERROR "configuration artifact identity: installed ${target}: ${location_absolute} != ${owned}")
            endif()
            file(REAL_PATH "${location_absolute}" location_real)
            file(REAL_PATH "${owned}" owned_real)
            cmake_path(IS_PREFIX prefix_real "${location_real}" NORMALIZE under_prefix)
            if(NOT under_prefix OR NOT location_real STREQUAL owned_real)
                message(FATAL_ERROR "Installed artifact escapes its owned prefix: ${target}/${location_real}")
            endif()
        elseif(NOT location STREQUAL expected_${id})
            message(FATAL_ERROR "configuration artifact identity: ${target}/${config}: ${location} != ${expected_${id}}")
        endif()
    endforeach()
endforeach()
if(package_name STREQUAL "component_test")
    include("${MOQ_HELPER_SOURCE}/interface-expectations.cmake")
    get_target_property(managed_kind moq::adapter-msquic-managed TYPE)
    get_target_property(managed_links moq::adapter-msquic-managed INTERFACE_LINK_LIBRARIES)
    if(NOT managed_kind STREQUAL "INTERFACE_LIBRARY" OR
       NOT managed_links STREQUAL "moq::adapter-msquic")
        message(FATAL_ERROR "Artifact-free managed component changed")
    endif()
    foreach(target IN LISTS targets)
        string(MAKE_C_IDENTIFIER "${target}" id)
        get_target_property(dirs "${target}" INTERFACE_LINK_DIRECTORIES)
        get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
        # Empty list elements have no usage meaning. Keep LINK_ONLY intact.
        list(FILTER links EXCLUDE REGEX "^$")
        if(MOQ_INSTALLED)
            set(want_links "${install_links_${id}}")
            set(want_dirs "${MOQ_INSTALL_PREFIX}/${install_dirs_relative_${id}}")
        else()
            set(want_links "${build_links_${id}}")
            set(want_dirs "${build_dirs_${id}}")
        endif()
        if(NOT "${dirs}" STREQUAL "${want_dirs}" OR NOT "${links}" STREQUAL "${want_links}")
            message(FATAL_ERROR "Component interface identity changed: ${target}: dirs=[${dirs}], links=[${links}]")
        endif()
        get_target_property(kind "${target}" TYPE)
        foreach(config IN LISTS MOQ_CONFIGS)
            string(TOUPPER "${config}" upper)
            include("${MOQ_EXPECTED}/expected-${config}.cmake")
            if(kind STREQUAL "SHARED_LIBRARY" AND DEFINED build_identity_${id})
                get_target_property(identity "${target}" "IMPORTED_SONAME_${upper}")
                if(MOQ_INSTALLED)
                    set(want_identity "${install_identity_${id}}")
                else()
                    set(want_identity "${build_identity_${id}}")
                endif()
                if(NOT identity STREQUAL want_identity)
                    message(FATAL_ERROR "Source shared identity changed: ${target}/${config}: ${identity} != ${want_identity}")
                endif()
            elseif(kind STREQUAL "STATIC_LIBRARY" AND own_cxx_${id} STREQUAL "adapter-mvfst")
                get_target_property(languages "${target}" "IMPORTED_LINK_INTERFACE_LANGUAGES_${upper}")
                if(NOT "CXX" IN_LIST languages)
                    message(FATAL_ERROR "Own-C++ static component lost its source language: ${target}/${config}")
                endif()
            endif()
        endforeach()
    endforeach()
    foreach(config IN LISTS MOQ_CONFIGS)
        string(TOUPPER "${config}" upper)
        get_target_property(no_soname probe::no_soname "IMPORTED_NO_SONAME_${upper}")
        if(NOT no_soname)
            message(FATAL_ERROR "Native reference lost source NO_SONAME/${config}")
        endif()
    endforeach()
endif()
set(smoke_components "${components}")
if(MOQ_CONSUMER_CASE STREQUAL "service-only")
    set(smoke_components service)
endif()
foreach(component IN LISTS smoke_components)
    string(MAKE_C_IDENTIFIER "${component}" id)
    foreach(lang c cxx)
        if(package_name STREQUAL "libmoq")
            if(lang STREQUAL "c")
                set(source "${CMAKE_CURRENT_SOURCE_DIR}/main.c")
            else()
                set(source "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
            endif()
        else()
            if(lang STREQUAL "c")
                set(source "${MOQ_HELPER_SOURCE}/main.c")
            else()
                set(source "${MOQ_HELPER_SOURCE}/main.cpp")
            endif()
        endif()
        add_executable("smoke_${id}_${lang}" "${source}")
        target_link_libraries("smoke_${id}_${lang}" PRIVATE "moq::${component}")
        target_compile_definitions("smoke_${id}_${lang}" PRIVATE "SMOKE_${id}=1")
        if(MOQ_CONSUMER_CASE STREQUAL "empty-release")
            target_compile_definitions("smoke_${id}_${lang}" PRIVATE PROBE_EXPECTED=2)
        endif()
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
            target_compile_options("smoke_${id}_${lang}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
        list(APPEND executables "smoke_${id}_${lang}")
    endforeach()
endforeach()
set(metadata "")
foreach(target IN LISTS targets)
    string(MAKE_C_IDENTIFIER "${target}" id)
    string(APPEND metadata "set(actual_${id} [==[$<TARGET_FILE:${target}>]==])\n")
endforeach()
string(APPEND metadata "set(executables [==[${executables}]==])\n")
foreach(exe IN LISTS executables)
    string(APPEND metadata "set(path_${exe} [==[$<TARGET_FILE:${exe}>]==])\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/actual-$<IF:$<BOOL:$<CONFIG>>,$<CONFIG>,NOCONFIG>.cmake"
    CONTENT "${metadata}")
