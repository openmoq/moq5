# Resolves the stand-in picoquic through libmoq's FindPicoquic in
# source-tree mode, then records how picoquic-core and picohttp-core name
# OpenSSL: "targets" (only through OpenSSL::SSL / OpenSSL::Crypto), "raw"
# (only through the artifacts' paths) or "mixed".
#   MOQ_TEST_PRE_ADD  add the stand-in before find_package, so its targets
#                     are pre-existing ones.
cmake_minimum_required(VERSION 3.20)
project(moq_picoquic_openssl_identity LANGUAGES C)
list(APPEND CMAKE_MODULE_PATH "${MOQ_TEST_MODULE_DIR}")
if(MOQ_TEST_PRE_ADD)
    add_subdirectory("${MOQ_PICOQUIC_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/pre-picoquic")
endif()
find_package(Picoquic REQUIRED)
find_package(OpenSSL REQUIRED)
set(raw "")
foreach(imp IN ITEMS OpenSSL::SSL OpenSSL::Crypto)
    get_target_property(loc ${imp} IMPORTED_LOCATION)
    list(APPEND raw "${loc}")
endforeach()
set(record "")
foreach(tgt IN ITEMS picoquic-core picohttp-core)
    get_target_property(items ${tgt} INTERFACE_LINK_LIBRARIES)
    set(n_raw 0)
    set(n_targets 0)
    foreach(item IN LISTS items)
        if(item MATCHES "^\\$<LINK_ONLY:([^<>]+)>$")
            set(item "${CMAKE_MATCH_1}")
        endif()
        if(item IN_LIST raw)
            math(EXPR n_raw "${n_raw} + 1")
        elseif(item STREQUAL "OpenSSL::SSL" OR item STREQUAL "OpenSSL::Crypto")
            math(EXPR n_targets "${n_targets} + 1")
        endif()
    endforeach()
    if(n_raw EQUAL 0 AND n_targets EQUAL 2)
        set(form targets)
    elseif(n_targets EQUAL 0 AND n_raw EQUAL 2)
        set(form raw)
    else()
        set(form "mixed(raw=${n_raw},targets=${n_targets})")
    endif()
    string(APPEND record "${tgt}=${form}\n")
endforeach()
file(WRITE "${MOQ_TEST_RESULT}" "${record}")
