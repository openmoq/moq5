cmake_minimum_required(VERSION 3.20)
project(moq_cold_component_consumer LANGUAGES C CXX)
if(TARGET Threads::Threads OR TARGET moq::service OR TARGET moq::adapter-msquic-managed OR
   TARGET msquic::msquic OR TARGET wtq::wtquic)
    message(FATAL_ERROR "Cold consumer scope was prepopulated; it cannot discriminate discovery")
endif()
if(EXPECT_THREADS STREQUAL "forbidden")
    # A real negative control: any unwanted required discovery is a CMake error.
    set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)
endif()
# No find_package(Threads), service find or direct managed-MsQuic find here.
find_package(libmoq CONFIG REQUIRED COMPONENTS ${MOQ_COLD_COMPONENT})
find_package(libmoq CONFIG REQUIRED COMPONENTS ${MOQ_COLD_COMPONENT})
if(TARGET moq::service OR TARGET moq::adapter-msquic-managed)
    message(FATAL_ERROR "Cold component-only lookup autoloaded a masking facade")
endif()
if(EXPECT_THREADS STREQUAL "forbidden" AND TARGET Threads::Threads)
    message(FATAL_ERROR "An unmanaged/shared attach consumer gained Threads discovery")
endif()
include("${EXPECTED_FILE}")
get_target_property(kind "moq::${MOQ_COLD_COMPONENT}" TYPE)
get_target_property(location "moq::${MOQ_COLD_COMPONENT}" IMPORTED_LOCATION_DEBUG)
if(NOT kind STREQUAL expected_kind OR NOT location STREQUAL expected)
    message(FATAL_ERROR "Cold component imported artifact identity changed: ${kind}/${location}")
endif()
get_target_property(links "moq::${MOQ_COLD_COMPONENT}" INTERFACE_LINK_LIBRARIES)
if(EXPECT_THREADS STREQUAL "required" AND NOT links MATCHES "Threads::Threads")
    message(FATAL_ERROR "Producer fixture did not expose the actual static Threads edge")
endif()
foreach(lang c cxx)
    if(lang STREQUAL "c")
        set(source main.c)
    else()
        set(source main.cpp)
    endif()
    add_executable("cold_${lang}" "${source}")
    target_link_libraries("cold_${lang}" PRIVATE "moq::${MOQ_COLD_COMPONENT}")
    if(MOQ_COLD_COMPONENT STREQUAL "adapter-msquic")
        target_compile_definitions("cold_${lang}" PRIVATE COLD_MSQUIC=1)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
        target_compile_options("cold_${lang}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
# Let native CMake generation diagnose an unresolved Threads::Threads edge.
# Inspect the generated target information only after configuration succeeds.
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/cold-import.cmake" CONTENT
    "set(actual [==[$<TARGET_FILE:moq::${MOQ_COLD_COMPONENT}>]==])\nset(threads_found $<TARGET_EXISTS:Threads::Threads>)\n")
