cmake_minimum_required(VERSION 3.20)
project(relay_consumer LANGUAGES C CXX)
set(RELAY_COMPONENT relay CACHE STRING "Requested relay component")
find_package(libmoq CONFIG REQUIRED COMPONENTS ${RELAY_COMPONENT})
foreach(lang IN ITEMS c cxx)
    add_executable(consumer_${lang} main.${lang})
    target_link_libraries(consumer_${lang} PRIVATE moq::${RELAY_COMPONENT})
    target_compile_features(consumer_${lang} PRIVATE c_std_11 cxx_std_11)
    if(RELAY_COMPONENT STREQUAL "relay")
        target_compile_definitions(consumer_${lang} PRIVATE TEST_RUNTIME)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(consumer_${lang} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
if(RELAY_COMPONENT STREQUAL "relay-core")
    if(TARGET moq::relay OR TARGET Threads::Threads)
        message(FATAL_ERROR "Engine consumer acquired runtime or thread dependency")
    endif()
endif()
