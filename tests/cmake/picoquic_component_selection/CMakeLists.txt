cmake_minimum_required(VERSION 3.20)
project(moq_component_selection LANGUAGES NONE)

# A configure-only SDK/provider model. The installed package config and
# FindPicoquic are production source; only exports and external dependencies
# are fakes. No compiler, provider download, archive or socket is involved.
include(CMakePackageConfigHelpers)
set(sdk "${CMAKE_CURRENT_BINARY_DIR}/sdk")
set(provider "${CMAKE_CURRENT_BINARY_DIR}/provider")
file(MAKE_DIRECTORY "${sdk}" "${provider}" "${provider}/include" "${provider}/lib")
set(MOQ_ADAPTER_PICOQUIC_NEEDS_OPENSSL OFF)
set(MOQ_BUILD_RELAY_CORE OFF)
set(MOQ_BUILD_RELAY_RUNTIME OFF)
configure_package_config_file("${MOQ_SOURCE}/cmake/libmoqConfig.cmake.in"
    "${sdk}/libmoqConfig.cmake" INSTALL_DESTINATION lib/cmake/libmoq)
configure_file("${MOQ_SOURCE}/cmake/libmoqRelayComponents.cmake.in"
    "${sdk}/libmoqRelayComponents.cmake" @ONLY)
foreach(module FindPicoquic MoqWtquicApi)
    configure_file("${MOQ_SOURCE}/cmake/${module}.cmake" "${sdk}/${module}.cmake" COPYONLY)
endforeach()
function(export_target file target deps)
    file(WRITE "${sdk}/${file}.cmake"
        "if(NOT TARGET ${target})\n  add_library(${target} INTERFACE IMPORTED)\n  set_property(TARGET ${target} PROPERTY INTERFACE_LINK_LIBRARIES \"${deps}\")\nendif()\n")
endfunction()
export_target(libmoqTargets moq::core "")
export_target(libmoqAdapterTargets moq::adapter-picoquic "moq::core;picoquic::picoquic-core")
export_target(libmoqAdapterThreadedTargets moq::adapter-picoquic-threaded "moq::adapter-picoquic;Threads::Threads")
export_target(libmoqPicoWtAdapterTargets moq::adapter-pico-wt "moq::core;picoquic::picohttp-core")
export_target(libmoqPicoWtManagedTargets moq::adapter-pico-wt-managed "moq::adapter-pico-wt;moq::adapter-picoquic;Threads::Threads")
set(service_deps "moq::adapter-picoquic-threaded;moq::adapter-pico-wt-managed;Threads::Threads")
if(MODEL_CASE STREQUAL "service-raw")
    file(REMOVE "${sdk}/libmoqPicoWtManagedTargets.cmake")
    set(service_deps "moq::adapter-picoquic-threaded;Threads::Threads")
elseif(MODEL_CASE STREQUAL "optional-missing-wt")
    file(REMOVE "${sdk}/libmoqPicoWtManagedTargets.cmake" "${sdk}/libmoqPicoWtAdapterTargets.cmake")
endif()
export_target(libmoqServiceTargets moq::service "${service_deps}")
file(WRITE "${sdk}/FindThreads.cmake" [=[
set(Threads_FOUND TRUE)
if(NOT TARGET Threads::Threads)
    add_library(Threads::Threads INTERFACE IMPORTED)
endif()
]=])
list(PREPEND CMAKE_MODULE_PATH "${sdk}")
set(PTLS_INCLUDE_DIR "${provider}/include")
set(MOQ_PTLS_LIB_DIR "${provider}/lib")
file(WRITE "${PTLS_INCLUDE_DIR}/picotls.h" "")
set(MODEL_HTTP ON)
if(MODEL_CASE MATCHES "^(core|raw|threaded|service-raw|optional-missing-wt|missing-wt|missing-service)$")
    set(MODEL_HTTP OFF)
endif()
set(picoquic_DIR "${provider}")
set(MODEL_PROVIDER_LOG "${CMAKE_CURRENT_BINARY_DIR}/provider.log")
file(WRITE "${MODEL_PROVIDER_LOG}" "")
file(WRITE "${provider}/picoquic-config.cmake" [=[
file(APPEND "${MODEL_PROVIDER_LOG}" "${_MOQ_PQ_NEED_HTTP}\n")
if(NOT TARGET picoquic::picoquic-core)
    add_library(picoquic::picoquic-core INTERFACE IMPORTED)
endif()
if(MODEL_HTTP AND _MOQ_PQ_NEED_HTTP AND NOT TARGET picoquic::picohttp-core)
    add_library(picoquic::picohttp-core INTERFACE IMPORTED)
endif()
]=])

macro(find_model expected)
    find_package(libmoq REQUIRED CONFIG PATHS "${sdk}" NO_DEFAULT_PATH ${ARGN})
    if(NOT _LIBMOQ_PICOQUIC_NEEDS_HTTP STREQUAL "${expected}")
        message(FATAL_ERROR "${MODEL_CASE}: HTTP hint '${_LIBMOQ_PICOQUIC_NEEDS_HTTP}', expected '${expected}'")
    endif()
endmacro()
if(MODEL_CASE STREQUAL "core")
    find_model(OFF)
elseif(MODEL_CASE STREQUAL "raw")
    find_model(OFF COMPONENTS adapter-picoquic)
elseif(MODEL_CASE STREQUAL "threaded")
    find_model(OFF COMPONENTS adapter-picoquic-threaded)
elseif(MODEL_CASE STREQUAL "optional-missing-wt")
    find_model(OFF COMPONENTS adapter-picoquic OPTIONAL_COMPONENTS adapter-pico-wt)
    if(libmoq_adapter-pico-wt_FOUND)
        message(FATAL_ERROR "Absent optional WT component reported found")
    endif()
elseif(MODEL_CASE STREQUAL "wt" OR MODEL_CASE STREQUAL "missing-wt")
    find_model(ON COMPONENTS adapter-pico-wt)
elseif(MODEL_CASE STREQUAL "managed")
    find_model(ON COMPONENTS adapter-pico-wt-managed)
elseif(MODEL_CASE STREQUAL "raw-wt")
    find_model(ON COMPONENTS adapter-picoquic adapter-pico-wt)
elseif(MODEL_CASE STREQUAL "wt-raw")
    find_model(ON COMPONENTS adapter-pico-wt adapter-picoquic)
elseif(MODEL_CASE STREQUAL "service" OR MODEL_CASE STREQUAL "missing-service")
    find_model(ON COMPONENTS service)
elseif(MODEL_CASE STREQUAL "service-raw")
    find_model(OFF COMPONENTS service)
elseif(MODEL_CASE STREQUAL "raw-then-wt")
    find_model(OFF COMPONENTS adapter-picoquic)
    if(TARGET picoquic::picohttp-core)
        message(FATAL_ERROR "Raw-only lookup unexpectedly loaded HTTP")
    endif()
    find_model(ON COMPONENTS adapter-pico-wt-managed)
elseif(MODEL_CASE STREQUAL "wt-then-raw")
    find_model(ON COMPONENTS adapter-pico-wt-managed)
    find_model(OFF COMPONENTS adapter-picoquic)
else()
    message(FATAL_ERROR "Unknown model case '${MODEL_CASE}'")
endif()

# Imported INTERFACE targets are lazy; assert their declared closure explicitly
# so an unvisited missing dependency cannot masquerade as configure success.
foreach(target moq::core moq::adapter-picoquic moq::adapter-picoquic-threaded
        moq::adapter-pico-wt moq::adapter-pico-wt-managed moq::service)
    if(TARGET ${target})
        get_target_property(deps ${target} INTERFACE_LINK_LIBRARIES)
        foreach(dep IN LISTS deps)
            if(NOT TARGET ${dep})
                message(FATAL_ERROR "${target} has unresolved dependency ${dep}")
            endif()
        endforeach()
    endif()
endforeach()
file(STRINGS "${MODEL_PROVIDER_LOG}" calls)
if(MODEL_CASE STREQUAL "core")
    if(calls OR TARGET picoquic::picoquic-core)
        message(FATAL_ERROR "Core-only consumer looked up a transport provider")
    endif()
elseif(MODEL_CASE STREQUAL "raw-then-wt")
    if(NOT calls STREQUAL "OFF;ON")
        message(FATAL_ERROR "Repeated find did not upgrade its provider requirement: '${calls}'")
    endif()
elseif(MODEL_HTTP)
    if(NOT calls STREQUAL "ON" OR NOT TARGET picoquic::picohttp-core)
        message(FATAL_ERROR "Required HTTP missing from first provider lookup: '${calls}'")
    endif()
else()
    if(NOT calls STREQUAL "OFF" OR TARGET picoquic::picohttp-core)
        message(FATAL_ERROR "Raw closure imposed HTTP: '${calls}'")
    endif()
endif()
