cmake_minimum_required(VERSION 3.20)
project(moq_package_embedding_consumer LANGUAGES C CXX)
# The caller owns foreign discovery; LibMoQ has no foreign-loader selector.
find_package(fixture_provider CONFIG REQUIRED)
find_package(libmoq CONFIG REQUIRED COMPONENTS service relay)
find_package(libmoq CONFIG REQUIRED COMPONENTS service relay)
foreach(lang c cxx)
    if(lang STREQUAL "c")
        set(source "${FIXTURE_SOURCE}/consumer.c")
    else()
        set(source "${FIXTURE_SOURCE}/consumer.cpp")
    endif()
    add_executable("embedded_${lang}" "${source}")
    target_link_libraries("embedded_${lang}" PRIVATE moq::service)
    if(lang STREQUAL "cxx")
        target_link_libraries("embedded_${lang}" PRIVATE moq::cpp)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
        target_compile_options("embedded_${lang}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    string(APPEND metadata "set(executable_${lang} [==[$<TARGET_FILE:embedded_${lang}>]==])\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/embedding-consumer.cmake" CONTENT "${metadata}")
